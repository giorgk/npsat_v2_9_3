cmake_minimum_required(VERSION 3.10)
find_package(deal.II 9.3 REQUIRED HINTS ${DEAL_II_DIR})
deal_ii_initialize_cached_variables()
project(well_routing_tests CXX)
find_package(Boost REQUIRED COMPONENTS program_options)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
# Expose the real tracer to this integration harness without production hooks.
file(READ "${REPO}/npsat_trace.cpp" tracer)
string(REPLACE "private:" "public:" tracer "${tracer}")
string(REPLACE "int main(" "int tracer_application_main(" tracer "${tracer}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/tracer_under_test.cpp" CONTENT "${tracer}")
add_executable(well_routing_tests well_routing_tests.cpp)
target_include_directories(well_routing_tests PRIVATE "${REPO}" "${CMAKE_CURRENT_BINARY_DIR}")
deal_ii_setup_target(well_routing_tests)
target_link_libraries(well_routing_tests ${Boost_LIBRARIES})
